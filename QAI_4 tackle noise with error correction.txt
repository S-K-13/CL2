# Step 2: Import required libraries
import numpy as np
from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator
from qiskit.visualization import plot_histogram
from IPython.display import display


# Step 3: Define the quantum circuit
qc = QuantumCircuit(3, 3)

qc.h(0)
qc.cx(0, 1)
qc.cz(0, 2)

qc.measure([0, 1, 2], [0, 1, 2])


# Step 4: Display the quantum circuit
display(qc.draw(output='mpl'))

# Display circuit in text format
print(qc.draw())


# Step 5: Create the simulator
backend = AerSimulator()


# Step 6: Transpile and simulate the original circuit
transpiled_qc = transpile(qc, backend)

job = backend.run(transpiled_qc, shots=1000)

result = job.result()

counts = result.get_counts()


# Step 7: Create calibration circuits
calibration_circuits = []

for state in range(8):

    cal_circuit = QuantumCircuit(3, 3)

    # Prepare the required basis state
    for qubit in range(3):
        if (state >> qubit) & 1:
            cal_circuit.x(qubit)

    # Measure all three qubits
    cal_circuit.measure([0, 1, 2], [0, 1, 2])

    calibration_circuits.append(cal_circuit)


# Step 8: Run the calibration circuits
transpiled_calibrations = transpile(calibration_circuits, backend)

cal_job = backend.run(transpiled_calibrations, shots=1000)

cal_results = cal_job.result()


# Step 9: Create the measurement calibration matrix
calibration_matrix = np.zeros((8, 8))

for state in range(8):

    cal_counts = cal_results.get_counts(state)

    # Store the measurement probabilities
    for measured_state, count in cal_counts.items():

        measured_value = int(measured_state, 2)

        calibration_matrix[measured_value, state] = count / 1000


# Step 10: Apply measurement error mitigation
measured_probabilities = np.zeros(8)

for state, count in counts.items():

    measured_value = int(state, 2)

    measured_probabilities[measured_value] = count / 1000

# Correct the measurement probabilities
mitigated_probabilities = np.linalg.solve(
    calibration_matrix, measured_probabilities
)

# Remove negative probabilities
mitigated_probabilities = np.clip(mitigated_probabilities, 0, None)

# Normalize the probabilities
mitigated_probabilities = (
    mitigated_probabilities / np.sum(mitigated_probabilities)
)


# Step 11: Convert probabilities into counts
mitigated_counts = {}

for state in range(8):

    binary_state = format(state, '03b')

    mitigated_counts[binary_state] = mitigated_probabilities[state] * 1000


# Step 12: Print original counts
print("Original counts:")
print(counts)


# Step 13: Print mitigated counts
print("\nMitigated counts:")
print(mitigated_counts)


# Step 14: Display original and mitigated histograms
plot_histogram(
    [counts, mitigated_counts],
    legend=['Original', 'Mitigated']
)

## 1. Aim

## 3. Theory and Important Quantum Concepts

1. Quantum Noise:
   - Quantum noise refers to unwanted disturbances that affect quantum states and operations.
   - It can occur due to environmental interactions, faulty gates and measurement errors.

2. Quantum Error Correction (QEC):
   - QEC is a technique used to detect and correct errors in quantum information.
   - It uses multiple physical qubits to protect one logical qubit.

3. Measurement Error:
   - Measurement errors occur when a qubit is measured incorrectly.
   - For example, |0> may be incorrectly recorded as 1.

4. Error Mitigation:
   - Error mitigation reduces the effect of errors using calibration and mathematical correction.
   - Unlike error correction, it does not necessarily detect and correct errors during computation.

5. Calibration:
   - Calibration involves preparing known quantum states and measuring them.
   - The results are used to estimate the probability of incorrect measurements.

6. Calibration Matrix:
   - A calibration matrix represents the relationship between prepared states and measured outcomes.
   - It is used to correct measurement probabilities mathematically.

7. Quantum Measurement:
   - Measurement converts quantum states into classical bits.
   - For 3 qubits, there are 8 possible measurement outcomes.

   Total states = 2^3 = 8

## 6. Quantum Gates and Operations Used

### 1. Hadamard Gate (H)

- Creates superposition in a qubit.
- Converts |0> into (|0> + |1>) / √2.

### 2. CNOT Gate (CX)

- A controlled gate that flips the target qubit when the control qubit is 1.
- Used to create correlations between qubits.

### 3. Controlled-Z Gate (CZ)

- Applies a phase flip when both qubits are in state |1>.
- Used as a two-qubit controlled phase operation.

### 4. Measurement

- Converts quantum states into classical bits.
- Used to obtain measurement outcomes for calibration and analysis.
