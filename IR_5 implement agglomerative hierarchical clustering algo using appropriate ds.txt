# Step 2: Import required libraries

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.decomposition import PCA
from sklearn.cluster import AgglomerativeClustering
from sklearn.preprocessing import StandardScaler, normalize
from sklearn.metrics import silhouette_score

import scipy.cluster.hierarchy as shc


# Step 3: Load dataset directly using URL

url = "https://archive.ics.uci.edu/ml/machine-learning-databases/00292/Wholesale%20customers%20data.csv"

df = pd.read_csv(url)


# Step 4: Display and understand the dataset

print("First five records:")
print(df.head())

print("\nDataset Shape:", df.shape)

print("\nDataset Information:")
print(df.info())

print("\nColumn Names:")
print(df.columns)

print("\nMissing Values:")
print(df.isnull().sum())


# Step 5: Select required features

# Remove categorical columns Channel and Region
X = df.drop(["Channel", "Region"], axis=1)

# Handle missing values
X = X.ffill()


# Step 6: Preprocess the data

# Scale all features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Normalize the scaled data
X_normalized = normalize(X_scaled)

X_normalized = pd.DataFrame(X_normalized)


# Step 7: Reduce dimensions using PCA

pca = PCA(n_components=2)
X_principal = pca.fit_transform(X_normalized)

X_principal = pd.DataFrame(X_principal)
X_principal.columns = ["P1", "P2"]

print("\nPCA Result:")
print(X_principal.head())


# Step 8: Visualize the Dendrogram

plt.figure(figsize=(8, 8))
plt.title("Visualizing the Dendrogram")

dendrogram = shc.dendrogram(
    shc.linkage(X_principal, method="ward")
)

plt.show()


# Step 9: Perform Agglomerative Clustering

# K = 2
ac2 = AgglomerativeClustering(n_clusters=2)
labels2 = ac2.fit_predict(X_principal)

plt.figure(figsize=(6, 6))
plt.scatter(X_principal["P1"], X_principal["P2"], c=labels2, cmap="rainbow")
plt.title("Agglomerative Clustering (K=2)")
plt.show()


# K = 3
ac3 = AgglomerativeClustering(n_clusters=3)
labels3 = ac3.fit_predict(X_principal)

plt.figure(figsize=(6, 6))
plt.scatter(X_principal["P1"], X_principal["P2"], c=labels3, cmap="rainbow")
plt.title("Agglomerative Clustering (K=3)")
plt.show()


# K = 4
ac4 = AgglomerativeClustering(n_clusters=4)
labels4 = ac4.fit_predict(X_principal)

plt.figure(figsize=(6, 6))
plt.scatter(X_principal["P1"], X_principal["P2"], c=labels4, cmap="rainbow")
plt.title("Agglomerative Clustering (K=4)")
plt.show()


# K = 5
ac5 = AgglomerativeClustering(n_clusters=5)
labels5 = ac5.fit_predict(X_principal)

plt.figure(figsize=(6, 6))
plt.scatter(X_principal["P1"], X_principal["P2"], c=labels5, cmap="rainbow")
plt.title("Agglomerative Clustering (K=5)")
plt.show()


# K = 6
ac6 = AgglomerativeClustering(n_clusters=6)
labels6 = ac6.fit_predict(X_principal)

plt.figure(figsize=(6, 6))
plt.scatter(X_principal["P1"], X_principal["P2"], c=labels6, cmap="rainbow")
plt.title("Agglomerative Clustering (K=6)")
plt.show()


# Step 10: Calculate Silhouette Scores

k = [2, 3, 4, 5, 6]

silhouette_scores = []

silhouette_scores.append(silhouette_score(X_principal, labels2))
silhouette_scores.append(silhouette_score(X_principal, labels3))
silhouette_scores.append(silhouette_score(X_principal, labels4))
silhouette_scores.append(silhouette_score(X_principal, labels5))
silhouette_scores.append(silhouette_score(X_principal, labels6))


# Display Silhouette Scores

print("\nSilhouette Scores:")

for i in range(len(k)):
    print("K =", k[i], "Score =", silhouette_scores[i])


# Step 11: Plot Silhouette Scores

plt.bar(k, silhouette_scores)

plt.xlabel("Number of Clusters")
plt.ylabel("Silhouette Score")
plt.title("Comparison of Silhouette Scores")

plt.show()


# Find optimal number of clusters

optimal_k = k[np.argmax(silhouette_scores)]

print("\nOptimal Number of Clusters:", optimal_k)

## 3. Theory and Important Concepts

- Clustering: It is an unsupervised machine learning technique that groups similar data points into clusters.
- Hierarchical Clustering: It creates a hierarchy of clusters based on similarities between data points.
- Agglomerative Clustering: It is a bottom-up approach where every data point initially forms an individual cluster, and similar clusters are repeatedly merged.
- Bottom-Up Approach: Initially, each data point is treated as a separate cluster. Clusters are gradually combined until the required number of clusters is obtained.
- Dendrogram: It is a tree-like diagram that represents how clusters are merged at different distances.
- Ward Linkage: A linkage method that merges clusters by minimizing the increase in within-cluster variance.
- Euclidean Distance: Measures the straight-line distance between two data points.
- Feature Scaling: Converts numerical features into a comparable scale.
- Standardization: Transforms features to have approximately zero mean and unit standard deviation.
- Normalization: Scales each data sample to a unit norm so that its overall magnitude becomes 1.
- PCA (Principal Component Analysis): A dimensionality reduction technique that transforms data into fewer components while preserving as much variance as possible.
- Principal Components: New features created by PCA that represent the major patterns of variation in the original data.
- Silhouette Score: Measures how well a data point fits within its own cluster compared to other clusters.
- Optimal Number of Clusters: The number of clusters that provides the best clustering quality according to the selected evaluation metric.

### Working of Agglomerative Clustering

1. Initially, every data point is considered an individual cluster.
2. Calculate the distances between clusters.
3. Identify the two closest clusters.
4. Merge the closest clusters.
5. Repeat the process until the required number of clusters is obtained.

## 6. Important Formulas and Calculations

### 1. Euclidean Distance

d(x, y) = √Σ(xᵢ - yᵢ)²

- x and y represent two data points.
- xᵢ and yᵢ represent their corresponding feature values.
- It measures the distance between two data points.

### 2. Standardization

Z = (X - μ) / σ

- X represents the original feature value.
- μ represents the mean of the feature.
- σ represents the standard deviation.
- Z represents the standardized value.
- It transforms features to approximately zero mean and unit variance.

### 3. Normalization

X_normalized = X / ||X||

- X represents a data sample.
- ||X|| represents its Euclidean norm.
- It scales each sample to unit length.

### 4. Ward Linkage

Ward linkage merges the two clusters that produce the minimum increase in within-cluster sum of squares.

ΔE = (n₁ × n₂ / (n₁ + n₂)) × ||μ₁ - μ₂||²

- n₁ and n₂ represent the number of points in two clusters.
- μ₁ and μ₂ represent their respective cluster means.
- It selects the merge that causes the smallest increase in within-cluster variance.

### 5. Silhouette Score

S(i) = (b(i) - a(i)) / max(a(i), b(i))

- a(i) represents the average distance of a data point from other points in its own cluster.
- b(i) represents the minimum average distance from points in another cluster.
- The score ranges from -1 to +1.
- A higher score generally indicates better-separated and more compact clusters.

### 6. Optimal Number of Clusters

Optimal K = K corresponding to the maximum Silhouette Score.

- The program compares scores for K = 2, 3, 4, 5 and 6.
- `np.argmax()` identifies the highest score.
- The corresponding K is selected as the optimal number of clusters.
